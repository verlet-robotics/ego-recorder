bool keyframe = (frame.frame_number % 30 == 0);

    reinterpret_cast<const uint16_t*>(frame.depth_data.data()),

**IMPORTANT H.264 empty-vector handling:** The H.264 encoder may return an empty vector for some frames (when the encoder is buffering internally). When h264_data is empty, we still need to write the frame to .egorec for the depth data (and the Zdepth data depends on frame ordering for P-frames). Write the frame with h264_data.size() = 0 for the RGB component. The reader will handle this by feeding packets to the decoder in order and retrieving decoded frames when available.

Wait -- actually this creates a problem. If H.264 returns empty data for some frames, the frame-to-decoded-RGB mapping breaks. With max_b_frames=0, the H.264 encoder should return data for every input frame (no reordering needed). But there can still be one-frame latency. To handle this cleanly:
- Check if h264_data is empty. If it is, that means the encoder is buffering. For max_b_frames=0, this should only happen for the very first frame.
- Still write the frame block with rgb_compressed_size=0. The reader will handle missing RGB data by re-feeding to the H.264 decoder later.

Actually, with libx264, preset "fast", no B-frames, the first avcodec_send_frame may return EAGAIN with no output. After the second frame is sent, the first frame's encoded output is returned along with the second. This is fine -- write whatever the encoder returns per frame, even if empty. The reader side must maintain a stateful decoder anyway.

```cpp
int main(int argc, char* argv[]) {
    // Check for subcommands before cxxopts parsing
    if (argc >= 2) {
        std::string_view cmd = argv[1];
        if (cmd == "info") {
            // ego-recorder info <file.egorec> [<file2.egorec> ...]
            if (argc < 3) {
                fprintf(stderr, "Usage: ego-recorder info <file.egorec> [...]\n");
                return 1;
            }
            for (int i = 2; i < argc; ++i) {
                const char* path = argv[i];
                std::ifstream in(path, std::ios::binary);
                if (!in.is_open()) {
                    fprintf(stderr, "Error: cannot open '%s'\n", path);
                    continue;
                }
                FileHeader hdr;
                in.read(reinterpret_cast<char*>(&hdr), sizeof(hdr));
                if (!in.good() || std::memcmp(hdr.magic, "EGOREC", 6) != 0) {
                    fprintf(stderr, "Error: '%s' is not a valid .egorec file\n", path);
                    continue;
                }

                // Read footer for frame count and duration
                in.seekg(-static_cast<int>(sizeof(FileFooter)), std::ios::end);
                FileFooter footer;
                in.read(reinterpret_cast<char*>(&footer), sizeof(footer));
                bool has_footer = in.good() && footer.footer_magic == FOOTER_MAGIC;

                printf("File: %s\n", path);
                printf("  Format version: %d.%d\n", hdr.magic[6], hdr.magic[7]);
                printf("  Session: %s\n", hdr.session_name);
                printf("  Serial: %s\n", hdr.serial_number);
                printf("  USB: %s\n", hdr.usb_type);
                printf("  Resolution: %ux%u (depth), %ux%u (color)\n",
                       hdr.depth_width, hdr.depth_height,
                       hdr.color_width, hdr.color_height);
                printf("  Depth scale: %.6f\n", hdr.depth_scale);
                printf("  RGB codec: %s (%d)\n",
                       hdr.rgb_codec == 0 ? "raw" : hdr.rgb_codec == 1 ? "JPEG" : hdr.rgb_codec == 2 ? "H264" : "unknown",
                       hdr.rgb_codec);
                printf("  Depth codec: %s (%d)\n",
                       hdr.depth_codec == 0 ? "raw" : hdr.depth_codec == 1 ? "ZSTD" : hdr.depth_codec == 2 ? "Zdepth" : "unknown",
                       hdr.depth_codec);
                printf("  RGB quality/CRF: %d\n", hdr.rgb_quality);
                printf("  Depth intrinsics: fx=%.2f fy=%.2f ppx=%.2f ppy=%.2f\n",
                       hdr.depth_fx, hdr.depth_fy, hdr.depth_ppx, hdr.depth_ppy);
                printf("  Color intrinsics: fx=%.2f fy=%.2f ppx=%.2f ppy=%.2f\n",
                       hdr.color_fx, hdr.color_fy, hdr.color_ppx, hdr.color_ppy);
                printf("  IMU: %s\n", (hdr.flags & 0x01) ? "yes" : "no");
                if (has_footer) {
                    printf("  Frames: %" PRIu64 "\n", footer.total_frames);
                    printf("  Duration: %.2f s\n", footer.total_duration_us / 1e6);
                }
                printf("\n");
            }
            return 0;
        }
    }

```
